        try {
          script = document.createElement('script');

          script.onerror = (error) => { cleanup(); reject(error); };
          document.head.appendChild(script);
        } catch (error) {

          reject(error);

  // Set a manual timeout for dropped packets to CDN
  function loadScriptWithTimeout(src, timeout) {
    return new Promise((resolve, reject) => {
      let script;
      let timer;
      const cleanup = () => {
        clearTimeout(timer);
        document.removeEventListener('DOMContentLoaded', appendScript);
      };
      const appendScript = () => {
        document.removeEventListener('DOMContentLoaded', appendScript);
        try {
          script = document.createElement('script');
          script.src = src;
          script.onload = () => { cleanup(); resolve(); };
          script.onerror = (error) => { cleanup(); reject(error); };
          document.head.appendChild(script);
        } catch (error) {
          cleanup();
          reject(error);
        }
      };

      // Arm the timeout immediately, independent of DOMContentLoaded. A CDN asset
      // whose packets are dropped can block the parser so DOMContentLoaded never
      // fires. Starting the timer here guarantees the no-CDN fallback still runs.
      timer = setTimeout(() => {
        cleanup();
        reject(new Error('Script load timeout'));
      }, timeout);

      // Inject the entry bundle only after the document has been parsed, so a
      // warm-cache bundle can't execute before #app-root exists in the DOM and
      // crash app bootstrap with "Element must be provided to render React component".
      if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', appendScript, { once: true });
      } else {
        appendScript();
      }
